cmake_minimum_required(VERSION 3.16)
project(passport_preview C)
set(CMAKE_C_STANDARD 11)
set(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(LVGL "${REPO}/managed_components/lvgl__lvgl")
file(GLOB_RECURSE LV_SOURCES "${LVGL}/src/*.c")
add_library(lvgl_host STATIC ${LV_SOURCES})
target_include_directories(lvgl_host PUBLIC "${LVGL}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(lvgl_host PUBLIC LV_CONF_INCLUDE_SIMPLE LV_KCONFIG_IGNORE)
add_executable(passport_preview preview.c "${REPO}/main/passport_ui.c" "${REPO}/main/passport_nav.c" "${REPO}/main/passport_art.c")
target_include_directories(passport_preview PRIVATE "${REPO}/main")
target_link_libraries(passport_preview PRIVATE lvgl_host)
if(NOT MSVC)
  target_compile_options(passport_preview PRIVATE -Wall -Wextra -Werror)
endif()
